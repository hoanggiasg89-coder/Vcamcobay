#import <UIKit/UIKit.h>
#import <AVFoundation/AVFoundation.h>
#import <Foundation/Foundation.h>

// ================= CẤU HÌNH TOÀN CỤC =================
static AVCaptureSession *g_session = nil;
static BOOL g_isVCamActive = NO;
static dispatch_queue_t g_videoQueue = NULL;
static NSString *kVCamEnabledKey = @"VCamEnabled";
static NSString *kShowLogoKey = @"ShowLogoOnly";

// Biến quản lý UI nổi
static UIWindow *g_floatingWindow = nil;
static UIView *g_floatingButtonView = nil;
static BOOL g_isPanelOpen = NO;

// Delegate để bắt các frame video từ session gốc
@interface VCamVideoDelegate : NSObject <AVCaptureVideoDataOutputSampleBufferDelegate>
@end

@implementation VCamVideoDelegate

// --- HÀM TIỆN ÍCH: TẠO FRAME TỪ ẢNH LOGO ---
- (CMSampleBufferRef)createFrameFromLogo {
    // 1. Đường dẫn đến hình ảnh VCAMCOBAY.png trong Resources
    NSString *imagePath = [[NSBundle mainBundle] pathForResource:@"vcamcobay_logo" ofType:@"png"];
    if (!imagePath) return NULL;
    
    UIImage *logoImage = [UIImage imageWithContentsOfFile:imagePath];
    if (!logoImage) return NULL;
    
    // 2. Chuyển đổi UIImage sang CVPixelBuffer
    CVPixelBufferRef buffer = NULL;
    size_t width = logoImage.size.width;
    size_t height = logoImage.size.height;
    
    NSDictionary *pixelAttributes = @{
        (__bridge id)kCVPixelBufferCGImageCompatibilityKey : @YES,
        (__bridge id)kCVPixelBufferCGBitmapContextCompatibilityKey : @YES
    };
    
    CVReturn status = CVPixelBufferCreate(kCFAllocatorDefault, 
                                          (int)width, 
                                          (int)height, 
                                          kCVPixelFormatType_32BGRA, 
                                          (__bridge CFDictionaryRef)pixelAttributes, 
                                          &buffer);
                                          
    if (status != kCVReturnSuccess) return NULL;
    
    // 3. Vẽ hình ảnh lên Pixel Buffer
    CVPixelBufferLockBaseAddress(buffer, 0);
    void *baseAddress = CVPixelBufferGetBaseAddress(buffer);
    
    CGContextRef context = CGBitmapContextCreate(baseAddress, 
                                                 (size_t)width, 
                                                 (size_t)height, 
                                                 8, 
                                                 (size_t)(CVPixelBufferGetBytesPerRow(buffer)), 
                                                 CGColorSpaceCreateDeviceRGB(), 
                                                 kCGBitmapByteOrder32Little | kCGImageAlphaPremultipliedFirst);
    
    if (context) {
        CGContextClearRect(context, CGRectMake(0, 0, width, height));
        CGContextDrawImage(context, CGRectMake(0, 0, width, height), logoImage.CGImage);
        CGContextRelease(context);
    }
    
    CVPixelBufferUnlockBaseAddress(buffer, 0);
    
    // 4. Tạo CMSampleBuffer từ Pixel Buffer
    CMTime presentationTime = CMTimeMake(0, 30); // Frame rate 30fps
    CMSampleBufferRef sampleBuffer = NULL;
    
    CMVideoFormatDescriptionRef videoInfo = NULL;
    CMVideoFormatDescriptionCreateForImageBuffer(kCFAllocatorDefault, buffer, &videoInfo);
    
    CMSampleTimingInfo timing = {CMTimeMake(1, 30), presentationTime, kCMTimeInvalid};
    
    OSStatus result = CMSampleBufferCreateForImageBuffer(kCFAllocatorDefault, 
                                                         buffer, 
                                                         true, 
                                                         NULL, 
                                                         NULL, 
                                                         videoInfo, 
                                                         &timing, 
                                                         &sampleBuffer);
                                                         
    CFRelease(videoInfo);
    CFRelease(buffer);
    
    return sampleBuffer;
}

// --- HOOK CHÍNH: XỬ LÝ FRAME ---
- (void)captureOutput:(AVCaptureOutput *)output didOutputSampleBuffer:(CMSampleBufferRef)sampleBuffer fromConnection:(AVCaptureConnection *)connection {
    if (g_isVCamActive && g_session != nil) {
        BOOL showLogo = [[NSUserDefaults standardUserDefaults] boolForKey:kShowLogoKey];
        
        if (showLogo) {
            // Thay thế frame camera thật bằng frame từ logo
            CMSampleBufferRef logoBuffer = [self createFrameFromLogo];
            
            if (logoBuffer) {
                NSLog(@"[Vcamcobay] Injecting Logo Frame");
                
                // Tại đây, bạn cần gửi logoBuffer đi. 
                // Trong một tweak đơn giản, chúng ta thường thay thế connection hoặc 
                // gửi buffer này đến một output khác mà app đích đang lắng nghe.
                
                // Ví dụ: Gửi trực tiếp nếu có cơ chế hook sâu hơn
                // [self sendBuffer:logoBuffer toConnection:connection];
                
                CFRelease(sampleBuffer); // Giải phóng frame cũ
            }
        }
    }
}

@end

// ================= UI NỔI (FLOATING UI) =================
@interface VCamFloatingController : UIViewController
@end

@implementation VCamFloatingController

- (void)viewDidLoad {
    [super viewDidLoad];
    
    if (!g_floatingWindow) {
        g_floatingWindow = [[UIWindow alloc] initWithFrame:[UIScreen mainScreen].bounds];
        g_floatingWindow.windowLevel = UIWindowLevelAlert + 100;
        g_floatingWindow.backgroundColor = [UIColor clearColor];
        g_floatingWindow.hidden = NO;
        
        g_floatingButtonView = [[UIView alloc] initWithFrame:CGRectMake(0, 0, 60, 60)];
        g_floatingButtonView.layer.cornerRadius = 30;
        g_floatingButtonView.backgroundColor = [UIColor colorWithWhite:0 alpha:0.85];
        g_floatingButtonView.clipsToBounds = YES;
        g_floatingButtonView.layer.shadowColor = [UIColor blackColor].CGColor;
        g_floatingButtonView.layer.shadowOpacity = 0.5;
        g_floatingButtonView.layer.shadowRadius = 5;
        
        UIImageView *logoView = [[UIImageView alloc] initWithImage:[UIImage imageNamed:@"vcamcobay_logo"]];
        logoView.contentMode = UIViewContentModeScaleAspectFit;
        logoView.frame = g_floatingButtonView.bounds;
        [g_floatingButtonView addSubview:logoView];
        
        UIPanGestureRecognizer *panGesture = [[UIPanGestureRecognizer alloc] initWithTarget:self action:@selector(handleDrag:)];
        [g_floatingButtonView addGestureRecognizer:panGesture];
        
        UITapGestureRecognizer *tapGesture = [[UITapGestureRecognizer alloc] initWithTarget:self action:@selector(togglePanel:)];
        tapGesture.numberOfTapsRequired = 1;
        [g_floatingButtonView addGestureRecognizer:tapGesture];
        
        [self.view addSubview:g_floatingButtonView];
        
        NSUserDefaults *defaults = [NSUserDefaults standardUserDefaults];
        NSValue *savedPos = [defaults objectForKey:@"VCamSavedPosition"];
        if (savedPos) {
            g_floatingButtonView.center = [savedPos CGPointValue];
        } else {
            g_floatingButtonView.center = CGPointMake([UIScreen mainScreen].bounds.size.width - 50, 
                                                      [UIScreen mainScreen].bounds.size.height / 2);
        }
    }
}

- (void)handleDrag:(UIPanGestureRecognizer *)gesture {
    CGPoint translation = [gesture translationInView:self.view];
    g_floatingButtonView.center = CGPointMake(
        g_floatingButtonView.center.x + translation.x,
        g_floatingButtonView.center.y + translation.y
    );
    [gesture setTranslation:CGPointZero inView:self.view];
}

- (void)viewWillDisappear:(BOOL)animated {
    [super viewWillDisappear:animated];
    NSUserDefaults *defaults = [NSUserDefaults standardUserDefaults];
    [defaults setObject:[NSValue valueWithCGPoint:g_floatingButtonView.center] forKey:@"VCamSavedPosition"];
    [defaults synchronize];
}

- (void)togglePanel:(UITapGestureRecognizer *)sender {
    if (!g_isPanelOpen) {
        [self showControlPanel];
    } else {
        [self hideControlPanel];
    }
}

- (void)showControlPanel {
    g_isPanelOpen = YES;
    CGRect panelFrame = CGRectMake(20, 100, 250, 300); 
    UIView *panel = [[UIView alloc] initWithFrame:panelFrame];
    panel.backgroundColor = [UIColor colorWithWhite:0 alpha:0.95];
    panel.layer.cornerRadius = 15;
    panel.layer.borderWidth = 1;
    panel.layer.borderColor = [[UIColor cyanColor] CGColor];
    
    UIButton *btnLogo = [UIButton buttonWithType:UIButtonTypeSystem];
    btnLogo.frame = CGRectMake(20, 20, 210, 40);
    [btnLogo setTitle:@"🎥 Hiển thị Logo VCAM" forState:UIControlStateNormal];
    [btnLogo addTarget:self action:@selector(toggleLogoAction:) forControlEvents:UIControlEventTouchUpInside];
    [panel addSubview:btnLogo];
    
    UIButton *btnClose = [UIButton buttonWithType:UIButtonTypeSystem];
    btnClose.frame = CGRectMake(20, 70, 210, 40);
    [btnClose setTitle:@"❌ Đóng Bảng Điều Khiển" forState:UIControlStateNormal];
    [btnClose addTarget:self action:@selector(hidePanelManually) forControlEvents:UIControlEventTouchUpInside];
    [panel addSubview:btnClose];
    
    g_floatingButtonView.hidden = YES;
    [self.view addSubview:panel];
    self.currentPanel = panel;
}

- (void)hideControlPanel {
    g_isPanelOpen = NO;
    if (self.currentPanel) {
        [self.currentPanel removeFromSuperview];
        self.currentPanel = nil;
    }
    g_floatingButtonView.hidden = NO;
}

- (void)hidePanelManually {
    [self togglePanel:nil];
}

- (void)toggleLogoAction:(id)sender {
    BOOL current = [[NSUserDefaults standardUserDefaults] boolForKey:kShowLogoKey];
    [[NSUserDefaults standardUserDefaults] setBool:!current forKey:kShowLogoKey];
    NSLog(@"[Vcamcobay] Logo Mode changed to: %@", !current ? @"ON" : @"OFF");
}

@property (nonatomic, strong) UIView *currentPanel;

@end

%hook AVCaptureSession

- (void)startRunning {
    BOOL isEnabled = [[NSUserDefaults standardUserDefaults] boolForKey:kVCamEnabledKey];
    
    if (isEnabled && !g_isVCamActive) {
        NSLog(@"[Vcamcobay] Intercepting Camera Session Start");
        g_session = self;
        g_isVCamActive = YES;
        
        if (!g_videoQueue) {
            g_videoQueue = dispatch_queue_create("com.vcamcobay.video.process", DISPATCH_QUEUE_SERIAL);
        }
        
        %orig;
    } else {
        %orig;
    }
}

- (void)stopRunning {
    NSLog(@"[Vcamcobay] Stopping Camera Session");
    g_isVCamActive = NO;
    g_session = nil;
    %orig;
}

%end

%hook UIApplication

- (BOOL)application:(UIApplication *)application didFinishLaunchingWithOptions:(NSDictionary *)launchOptions {
    BOOL didLaunch = %orig;
    
    if (!g_floatingWindow) {
        VCamFloatingController *controller = [[VCamFloatingController alloc] init];
        g_floatingWindow.rootViewController = controller;
        [g_floatingWindow makeKeyAndVisible];
    }
    
    return didLaunch;
}

%end
