// ================= LỚP UI NỔI VỚI KHẢ NĂNG KÉO THẢ =================
@interface VCamFloatingController : UIViewController <UIResponder>
@end

@implementation VCamFloatingController

- (void)viewDidLoad {
    [super viewDidLoad];
    
    // 1. Tạo Window nổi nếu chưa có
    if (!g_floatingWindow) {
        g_floatingWindow = [[UIWindow alloc] initWithFrame:[UIScreen mainScreen].bounds];
        g_floatingWindow.windowLevel = UIWindowLevelAlert + 100;
        g_floatingWindow.backgroundColor = [UIColor clearColor];
        g_floatingWindow.hidden = NO;
        
        // 2. Tạo Nút Tròn
        g_floatingButtonView = [[UIView alloc] initWithFrame:CGRectMake(0, 0, 60, 60)];
        g_floatingButtonView.layer.cornerRadius = 30;
        g_floatingButtonView.backgroundColor = [UIColor colorWithWhite:0 alpha:0.85];
        g_floatingButtonView.clipsToBounds = YES;
        g_floatingButtonView.layer.shadowColor = [UIColor blackColor].CGColor;
        g_floatingButtonView.layer.shadowOpacity = 0.5;
        g_floatingButtonView.layer.shadowRadius = 5;
        
        // Thêm Logo vào nút
        UIImageView *logoView = [[UIImageView alloc] initWithImage:[UIImage imageNamed:@"vcamcobay_logo"]];
        logoView.contentMode = UIViewContentModeScaleAspectFit;
        logoView.frame = g_floatingButtonView.bounds;
        [g_floatingButtonView addSubview:logoView];
        
        // 3. Kích hoạt khả năng KÉO THẢ
        UIPanGestureRecognizer *panGesture = [[UIPanGestureRecognizer alloc] initWithTarget:self action:@selector(handleDrag:)];
        [g_floatingButtonView addGestureRecognizer:panGesture];
        
        // 4. Kích hoạt khả năng NHẤP ĐỂ MỞ BẢNG ĐIỀU KHIỂN
        UITapGestureRecognizer *tapGesture = [[UITapGestureRecognizer alloc] initWithTarget:self action:@selector(togglePanel:)];
        tapGesture.numberOfTapsRequired = 1;
        [g_floatingButtonView addGestureRecognizer:tapGesture];
        
        [self.view addSubview:g_floatingButtonView];
        
        // Load vị trí cũ từ bộ nhớ
        NSUserDefaults *defaults = [NSUserDefaults standardUserDefaults];
        NSValue *savedPos = [defaults objectForKey:@"VCamSavedPosition"];
        if (savedPos) {
            g_floatingButtonView.center = [savedPos CGPointValue];
        } else {
            // Vị trí mặc định: Góc phải giữa màn hình
            g_floatingButtonView.center = CGPointMake([UIScreen mainScreen].bounds.size.width - 50, 
                                                      [UIScreen mainScreen].bounds.size.height / 2);
        }
    }
}

// --- HÀM XỬ LÝ KÉO THẢ ---
- (void)handleDrag:(UIPanGestureRecognizer *)gesture {
    CGPoint translation = [gesture translationInView:self.view];
    
    // Cập nhật vị trí mới
    g_floatingButtonView.center = CGPointMake(
        g_floatingButtonView.center.x + translation.x,
        g_floatingButtonView.center.y + translation.y
    );
    
    // Reset translation để lần kéo tiếp theo không bị nhảy
    [gesture setTranslation:CGPointZero inView:self.view];
}

// --- HÀM LUÔN LƯU VỊ TRÍ SAU KHI KÉO HOẶC TẮT APP ---
- (void)viewWillDisappear:(BOOL)animated {
    [super viewWillDisappear:animated];
    NSUserDefaults *defaults = [NSUserDefaults standardUserDefaults];
    [defaults setObject:[NSValue valueWithCGPoint:g_floatingButtonView.center] forKey:@"VCamSavedPosition"];
    [defaults synchronize];
}

// --- HÀM CHUYỂN ĐỔI GIỮA NÚT TRÒN VÀ BẢNG ĐIỀU KHIỂN ---
- (void)togglePanel:(UITapGestureRecognizer *)sender {
    if (!g_isPanelOpen) {
        [self showControlPanel];
    } else {
        [self hideControlPanel];
    }
}

- (void)showControlPanel {
    g_isPanelOpen = YES;
    
    // Tạo khung chữ nhật nhỏ gọn
    CGRect panelFrame = CGRectMake(20, 100, 250, 300); 
    UIView *panel = [[UIView alloc] initWithFrame:panelFrame];
    panel.backgroundColor = [UIColor colorWithWhite:0 alpha:0.95];
    panel.layer.cornerRadius = 15;
    panel.layer.borderWidth = 1;
    panel.layer.borderColor = [[UIColor cyanColor] CGColor];
    
    // Nút bật/tắt Logo
    UIButton *btnLogo = [UIButton buttonWithType:UIButtonTypeSystem];
    btnLogo.frame = CGRectMake(20, 20, 210, 40);
    [btnLogo setTitle:@"🎥 Hiển thị Logo VCAM" forState:UIControlStateNormal];
    [btnLogo addTarget:self action:@selector(toggleLogoAction:) forControlEvents:UIControlEventTouchUpInside];
    [panel addSubview:btnLogo];
    
    // Nút đóng bảng điều khiển
    UIButton *btnClose = [UIButton buttonWithType:UIButtonTypeSystem];
    btnClose.frame = CGRectMake(20, 70, 210, 40);
    [btnClose setTitle:@"❌ Đóng Bảng Điều Khiển" forState:UIControlStateNormal];
    [btnClose addTarget:self action:@selector(hidePanelManually) forControlEvents:UIControlEventTouchUpInside];
    [panel addSubview:btnClose];
    
    // Ẩn nút tròn, hiện panel
    g_floatingButtonView.hidden = YES;
    [self.view addSubview:panel];
    self.currentPanel = panel;
}

- (void)hideControlPanel {
    g_isPanelOpen = NO;
    if (self.currentPanel) {
        [self.currentPanel removeFromSuperview];
        self.currentPanel = nil;
    }
    g_floatingButtonView.hidden = NO;
}

- (void)hidePanelManually {
    [self togglePanel:nil];
}

- (void)toggleLogoAction:(id)sender {
    BOOL current = [[NSUserDefaults standardUserDefaults] boolForKey:kShowLogoKey];
    [[NSUserDefaults standardUserDefaults] setBool:!current forKey:kShowLogoKey];
    NSLog(@"[Vcamcobay] Logo Mode changed to: %@", !current ? @"ON" : @"OFF");
}

@property (nonatomic, strong) UIView *currentPanel;

@end
